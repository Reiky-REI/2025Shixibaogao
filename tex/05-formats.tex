  [0em]
  {\normalfont\normalsize}
  {第\chinese{section}章\quad}
  {}
  {\titlerule*[0.5pc]{.}\contentspage}

	itlecontents{subsection}
  [2em]
  {\normalfont\normalsize}
  {\chinese{subsection}、\quad}
  {}
  {\titlerule*[0.5pc]{.}\contentspage}

	itlecontents{subsubsection}
  [3.5em]
  {\normalfont\normalsize}
  {\arabic{subsubsection}.\quad}
  {}
  {\titlerule*[0.5pc]{.}\contentspage}

	itlecontents{paragraph}
  [5em]
  {\normalfont\normalsize}
  {(\arabic{paragraph})\quad}
  {}
  {\titlerule*[0.5pc]{.}\contentspage}

% 标题格式
% 章：小三、居中、显示“第X章”
	itleformat{\section}
  {\centering\normalfont\zihao{-3}}
  {第\chinese{section}章}
  {1em}{}
	itlespacing{\section}{0pt}{1\baselineskip}{1\baselineskip}

% 节：四号、顶头
	itleformat{\subsection}
  {\normalfont\zihao{4}}
  {\chinese{subsection}、}
  {1em}{}
	itlespacing{\subsection}{0pt}{0.8\baselineskip}{0.6\baselineskip}

% 小节：小四
	itleformat{\subsubsection}
  {\normalfont\zihao{-4}}
  {\arabic{subsubsection}.}
  {0.8em}{}
	itlespacing{\subsubsection}{0pt}{0.7\baselineskip}{0.4\baselineskip}

% 段落：小四，单独成行（不再 run-in，避免 titlesec 报错）
	itleformat{\paragraph}
  {\normalfont\zihao{-4}}
  {(\arabic{paragraph})}
  {0.8em}{}
	itlespacing{\paragraph}{0pt}{0.6\baselineskip}{0.2\baselineskip}
